const http = require('node:http'); const FINN_URL = 'https://www.finn.no/recommerce/seller/553930947'; http.createServer(async (req, res) => { if (req.url === '/api/finn') { try { const response = await fetch(FINN_URL); if (!response.ok) { throw new Error(`FINN HTTP ${response.status}`); } const html = await response.text(); res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(html); } catch (error) { res.writeHead(502); res.end(error.message); } return; } res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(`

FINN Satıcı Bilgileri

Yükleniyor...
`); }).listen(3000, () => { console.log('http://localhost:3000'); });
